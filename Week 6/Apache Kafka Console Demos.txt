Lab 1: Multiple Consumers Demo
#create topic in producer
C:\kafka_2.13-3.9.1>bin\windows\kafka-topics.bat --create --topic group-demo --bootstrap-server localhost:9092 --partitions 3 --replication-factor 1

#start producer
C:\kafka_2.13-3.9.1>bin\windows\kafka-console-producer.bat --topic group-topic --bootstrap-server localhost:9092

#start consumer 1. In new terminal(símbolo del sistema)
C:\kafka_2.13-3.9.1>bin\windows\kafka-console-consumer.bat --topic group-topic --from-beginning --bootstrap-server localhost:9092

#start consumer 2. In new terminal(símbolo del sistema)
C:\kafka_2.13-3.9.1>bin\windows\kafka-console-consumer.bat --topic group-topic --from-beginning --bootstrap-server localhost:9092
